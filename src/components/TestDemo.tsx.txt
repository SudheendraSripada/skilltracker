'use client';

import { useState } from 'react';
import { createTest } from '@/actions/tests';

export default function TestDemo() {
  const [status, setStatus] = useState<string>('Ready to test');
  const [loading, setLoading] = useState(false);

  const handleCreateTest = async () => {
    setLoading(true);
    setStatus('Creating test...');

    try {
      // We pass a dummy topic_id (must be a valid UUID format)
      const response = await createTest({
        topic_id: '123e4567-e89b-12d3-a456-426614174000', 
        status: 'offered',
        total_questions: 5,
        max_score: 100,
      });

      if (response.success && response.data) {
        setStatus(`✅ Success! Created test with ID: ${response.data.id}`);
      } else {
        setStatus(`❌ Failed: ${response.error}`);
      }
    } catch (error) {
      setStatus(`❌ Error: ${String(error)}`);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="p-6 mt-10 max-w-md mx-auto bg-white rounded-xl shadow-md space-y-4 border border-gray-200">
      <h2 className="text-xl font-bold text-gray-800">Supabase Connection Test</h2>
      <button
        onClick={handleCreateTest}
        disabled={loading}
        className="w-full px-4 py-2 bg-blue-600 text-white font-semibold rounded hover:bg-blue-700 disabled:bg-gray-400 transition-colors"
      >
        {loading ? 'Creating...' : 'Create Blank Test'}
      </button>
      <p className="text-sm font-medium text-gray-600">{status}</p>
    </div>
  );
}
